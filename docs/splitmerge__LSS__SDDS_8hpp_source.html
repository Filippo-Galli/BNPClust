321 SplitMerge_LSS_SDDS(Data &d, utils_params &p, Likelihood &l, Process &pr,
322 bool shuffle)
323 : Sampler(d, l, pr), params(p), shuffle_bool(shuffle),
324 original_allocations(pr.old_allocations_view()), gen(rd()),
325 dis_real(0.0, 1.0), dis_int(0, 1) {
326
327 // Precompute row max distances for efficiency
328 row_max_distances.resize(data.get_n());
329 for (int i = 0; i < data.get_n(); ++i) {
330 row_max_distances[i] = params.D.row(i).maxCoeff();
331 }
332 const int n = data.get_n();
333
334 // Take all pairwise distances (i < j)
335 std::vector<double> pairwise_dists;
336 pairwise_dists.reserve(n * (n - 1) / 2);
337 for (int i = 0; i < n; ++i) {
338 for (int j = i + 1; j < n; ++j) {
339 pairwise_dists.push_back(params.D(i, j));
340 }
341 }
342
343 // Compute the 50th percentile of pairwise distances
344 if (!pairwise_dists.empty()) {
345 auto q_it = pairwise_dists.begin() +
346 static_cast<size_t>(pairwise_dists.size() * 0.50);
347 std::nth_element(pairwise_dists.begin(), q_it, pairwise_dists.end());
348 temperature = *q_it;
349 if (temperature < 1e-12)
350 temperature = 1.0;
351 } else {
352 temperature = 1.0;
353 }
354 };
